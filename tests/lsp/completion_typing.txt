use std::fs;

struct Point {
  pub x: i32;
  pub y: i32;

  pub fn sum(self) -> i32 {
    self.x + self.y
  }
}

class Counter {
  pub var count: i64;
  pub var label: String;

  pub fn new(self) {
    self.count = 0;
    self.label = "clicks";
  }

  pub fn origin(self) -> Point {
    Point { x: 0, y: 0 }
  }
}
